/* ========================================================= SEGNALA FURTO / RECUPERA BICI ========================================================= */ async function toggle(idValue){ const b = data.find( q => q.id === idValue ); if(!b){ alert( "Bici non trovata." ); return; } const nuovoStato = b.stolen !== true; const nuovoStatus = nuovoStato ? "stolen" : "active"; const conferma = nuovoStato ? confirm( "Vuoi segnalare questa bici come RUBATA?\n\n" + "Lo stato verrà aggiornato immediatamente su BIICODE." ) : confirm( "Vuoi segnalare questa bici come RECUPERATA?" ); if(!conferma){ return; } const button = document.getElementById( "toggleButton" ); if(button){ button.disabled = true; button.textContent = nuovoStato ? "SEGNALAZIONE IN CORSO..." : "RECUPERO IN CORSO..."; } try{ /* ========================================== AGGIORNAMENTO SUPABASE ========================================== */ const { data:updatedRows, error } = await db .from("bikes") .update({ stolen: nuovoStato, status: nuovoStatus }) .eq( "biicode_id", idValue ) .select(); if(error){ console.error( "Errore aggiornamento Supabase:", error ); alert( "Impossibile aggiornare la bici su Supabase.\n\n" + error.message ); if(button){ button.disabled = false; button.textContent = b.stolen ? "SEGNALA COME RECUPERATA" : "SEGNALA FURTO"; } return; } /* ========================================== CONTROLLO RISULTATO ========================================== */ if( !updatedRows || updatedRows.length === 0 ){ console.error( "Nessuna riga restituita dopo UPDATE" ); alert( "Supabase non ha aggiornato la bici.\n\n" + "Probabilmente devi controllare la policy UPDATE nella tabella bikes." ); if(button){ button.disabled = false; button.textContent = b.stolen ? "SEGNALA COME RECUPERATA" : "SEGNALA FURTO"; } return; } /* ========================================== USIAMO IL DATO RESTITUITO DA SUPABASE ========================================== */ const updatedBike = normalizeBike( updatedRows[0] ); const index = data.findIndex( q => q.id === idValue ); if(index !== -1){ data[index] = updatedBike; } save(); /* ========================================== VERIFICA ULTERIORE DIRETTAMENTE DA DATABASE ========================================== */ const { data:checkRows, error:checkError } = await db .from("bikes") .select("*") .eq( "biicode_id", idValue ) .limit(1); if( !checkError && checkRows && checkRows.length ){ const checkedBike = normalizeBike( checkRows[0] ); const checkedIndex = data.findIndex( q => q.id === idValue ); if(checkedIndex !== -1){ data[checkedIndex] = checkedBike; } save(); } /* ========================================== MESSAGGIO ========================================== */ alert( nuovoStato ? "🚨 FURTO SEGNALATO CORRETTAMENTE" : "🟢 BICI SEGNALATA COME RECUPERATA" ); /* ========================================== RICARICA DETTAGLIO ========================================== */ detail( idValue ); }catch(error){ console.error( "Errore collegamento:", error ); alert( "Errore di collegamento con Supabase.\n\n" + error.message ); if(button){ button.disabled = false; button.textContent = b.stolen ? "SEGNALA COME RECUPERATA" : "SEGNALA FURTO"; } } }
